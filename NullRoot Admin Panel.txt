-- ═══════════════════════════════════════════════════════════
-- RANK PARSER
-- ═══════════════════════════════════════════════════════════
local function nameToLevel(name)
    if not name then return nil end
    local l = tostring(name):lower()
    if l:find("owner") then return 5 end
    if l:find("headadmin") or l:find("superadmin") then return 4 end
    if l:find("admin") and not l:find("nonadmin") then return 3 end
    if l:find("mod") then return 2 end
    if l:find("vip") then return 1 end
    if l:find("nonadmin") or l:find("guest") then return 0 end
    return nil
end

local function parseRank(v)
    if v == nil then return nil end
    if type(v) == "number" then
        if v >= 0 and v <= 5 then return v end
        if v == 254 or v == 255 then return 5 end
    end
    if type(v) == "table" then
        if v.RankId then return tonumber(v.RankId) end
        if v.Level then return tonumber(v.Level) end
        if v.RankName then return nameToLevel(v.RankName) end
        if v.Rank then return type(v.Rank) == "number" and v.Rank or nameToLevel(v.Rank) end
        return nil
    end
    return nameToLevel(tostring(v))
end

local function updateUIRank()
    if NR.rankLabel then
        NR.rankLabel.Text = string.format("● %s (%d)", State.currentName, State.currentRank)
        if State.currentRank >= 4 then
            NR.rankLabel.TextColor3 = C_SUCCESS
        elseif State.currentRank >= 2 then
            NR.rankLabel.TextColor3 = C_WARN
        elseif State.currentRank >= 1 then
            NR.rankLabel.TextColor3 = C_ACCENT
        else
            NR.rankLabel.TextColor3 = C_TEXT_DIM
        end
    end
    if NR.rankStatus then
        NR.rankStatus.Text = string.format("Commands: %d | %s (%d) | src: %s",
            #State.commandButtons, State.currentName, State.currentRank,
            State.hitSource or "—")
    end
    if NR.rankSelectBtn then
        NR.rankSelectBtn.Text = "Rank: " .. State.currentName
    end
end

local function setAchieved(level, source)
    if not level or level <= State.achievedRank then return false end
    State.achievedRank = level
    State.achievedName = RANK_NAMES[level] or ("L" .. level)
    State.hitSource = source
    NR.addLog(string.format("*** ACHIEVED: %s (%d) via %s ***",
        State.achievedName, level, source))
    NR.notify("Rank Up!", State.achievedName .. " (" .. level .. ")", 5)
    
    if level > State.currentRank then
        State.currentRank = level
        State.currentName = State.achievedName
        updateUIRank()
        if NR.generateButtons then NR.generateButtons() end
    end
    return true
end

NR.setAchieved = setAchieved

local function setCurrent(level)
    if level > State.achievedRank then
        NR.addLog("cannot select rank above achieved (" .. State.achievedRank .. ")")
        return false
    end
    State.currentRank = level
    State.currentName = RANK_NAMES[level] or ("L" .. level)
    updateUIRank()
    if NR.generateButtons then NR.generateButtons() end
    return true
end

NR.setCurrent = setCurrent

-- ═══════════════════════════════════════════════════════════
-- DETECTION #1: RankChanged signal
-- ═══════════════════════════════════════════════════════════
local function hookRankChanged()
    if State.rankHooked then return end
    local hd = RS:FindFirstChild("HDAdminClient")
    local signals = hd and hd:FindFirstChild("Signals")
    local rc = signals and signals:FindFirstChild("RankChanged")
    if rc and rc:IsA("RemoteEvent") then
        State.rankHooked = true
        rc.OnClientEvent:Connect(function(newRank)
            local lvl = parseRank(newRank)
            if lvl then setAchieved(lvl, "RankChanged-signal") end
        end)
        NR.addLog("hooked RankChanged signal")
    end
end

-- ═══════════════════════════════════════════════════════════
-- DETECTION #2: RetrieveRankInfo
-- ═══════════════════════════════════════════════════════════
local function queryRetrieveRankInfo()
    local hd = RS:FindFirstChild("HDAdminClient")
    local signals = hd and hd:FindFirstChild("Signals")
    local rri = signals and signals:FindFirstChild("RetrieveRankInfo")
    if not rri or rri.ClassName ~= "RemoteFunction" then return end
    
    local sigs = {{LP}, {LP.UserId}, {}, {LP.Name}}
    for _, args in ipairs(sigs) do
        local ok, result = pcall(function()
            return rri:InvokeServer(table.unpack(args))
        end)
        if ok and result then
            local lvl = parseRank(result)
            if lvl then
                setAchieved(lvl, "RetrieveRankInfo")
                return
            end
        end
    end
end

-- ═══════════════════════════════════════════════════════════
-- DETECTION #3: Attributes
-- ═══════════════════════════════════════════════════════════
local function scanAttributes()
    for k, v in pairs(LP:GetAttributes()) do
        local l = k:lower()
        if l:find("rank") or l:find("admin") or l:find("hd") then
            local lvl = parseRank(v)
            if lvl then setAchieved(lvl, "attr:" .. k) end
        end
    end
end

-- ═══════════════════════════════════════════════════════════
-- DETECTION #4: HD Admin UI header
-- ═══════════════════════════════════════════════════════════
local function scanUIHeader()
    for _, gui in ipairs(parent:GetChildren()) do
        if gui:IsA("ScreenGui") then
            local gname = gui.Name:lower()
            if gname:find("hd") or gname:find("admin") or gname:find("command") then
                for _, v in ipairs(gui:GetDescendants()) do
                    if v:IsA("TextLabel") and v.Text then
                        local lvl = parseRank(v.Text)
                        if lvl then setAchieved(lvl, "ui-header") end
                    end
                end
            end
        end
    end
end

-- ═══════════════════════════════════════════════════════════
-- DETECTION #5: Chat log scan
-- ═══════════════════════════════════════════════════════════
local function scanChatLog()
    for _, gui in ipairs(parent:GetChildren()) do
        if gui:IsA("ScreenGui") and gui.Name:lower():find("chat") then
            for _, v in ipairs(gui:GetDescendants()) do
                if v:IsA("TextLabel") and v.Text then
                    local lvl = parseRank(v.Text)
                    if lvl then setAchieved(lvl, "chat-log") end
                end
            end
        end
    end
end

-- ═══════════════════════════════════════════════════════════
-- DETECT RUNNER
-- ═══════════════════════════════════════════════════════════
local function detect()
    pcall(hookRankChanged)
    pcall(queryRetrieveRankInfo)
    pcall(scanAttributes)
    pcall(scanUIHeader)
    pcall(scanChatLog)
end

NR.detect = detect

-- ═══════════════════════════════════════════════════════════
-- CHAT EXECUTION
-- ═══════════════════════════════════════════════════════════
function NR.sendChat(cmd)
    if cmd == "" then return false end
    local sent = false
    
    pcall(function()
        if TextChatService.ChatVersion == Enum.ChatVersion.TextChatService then
            local ch = TextChatService:FindFirstChild("TextChannels")
            local gen = ch and (ch:FindFirstChild("RBXGeneral") or ch:FindFirstChildWhichIsA("TextChannel"))
            if gen then gen:SendAsync(cmd); sent = true end
        end
    end)
    
    if not sent then
        pcall(function()
            local ev = RS:FindFirstChild("DefaultChatSystemChatEvents")
            local sm = ev and ev:FindFirstChild("SayMessageRequest")
            if sm then sm:FireServer(cmd, "All"); sent = true end
        end)
    end
    
    return sent
end

function NR.executeCommand(cmd)
    if cmd == "" then return end
    cmd = cmd:gsub("%s+me%s*$", ""):gsub("%s+self%s*$", "")
    if cmd == "" then return end
    NR.addLog("> " .. cmd)
    NR.sendChat(cmd)
end

-- ═══════════════════════════════════════════════════════════
-- REMOTE COLLECTION
-- ═══════════════════════════════════════════════════════════
local function collectRemotes()
    local suspects, nonHD = {}, {}
    local SUSPECT_NAMES = {
        "purchase","buy","admin","mod","rank","grant","unlock",
        "claim","coin","shop","prompt","store","set","request",
        "make","promote","give",
    }
    for _, v in ipairs(game:GetDescendants()) do
        if v:IsA("RemoteEvent") then
            local path = v:GetFullName()
            if not path:find("RobloxReplicatedStorage")
               and not path:find("DefaultChatSystemChatEvents")
               and v.Name ~= "__FUNCTION" then
                if not path:find("HDAdmin") then
                    table.insert(nonHD, v)
                    local n = v.Name:lower()
                    for _, c in ipairs(SUSPECT_NAMES) do
                        if n:find(c) then table.insert(suspects, v); break end
                    end
                end
            end
        end
    end
    return { suspects = suspects, nonHD = nonHD }
end

local function fireParallel(remotes, args)
    local r = {}
    for _, v in ipairs(args) do
        if v == nil then table.insert(r, LP.UserId)
        else table.insert(r, v) end
    end
    local isTbl = type(r[1]) == "table"
    for i = 1, #remotes, CONFIG.BATCH_SIZE do
        local stop = math.min(i + CONFIG.BATCH_SIZE - 1, #remotes)
        for j = i, stop do
            local remote = remotes[j]
            task.spawn(function()
                pcall(function()
                    if isTbl then remote:FireServer(r[1])
                    else remote:FireServer(table.unpack(r)) end
                end)
            end)
        end
        task.wait(CONFIG.BATCH_WAIT)
    end
end

local function raceFire(remotes, args)
    local r = {}
    for _, v in ipairs(args) do
        if v == nil then table.insert(r, LP.UserId)
        else table.insert(r, v) end
    end
    for _, remote in ipairs(remotes) do
        for i = 1, CONFIG.RACE_COUNT do
            task.spawn(function()
                pcall(function() remote:FireServer(table.unpack(r)) end)
            end)
        end
    end
end

-- ═══════════════════════════════════════════════════════════
-- NOTIFICATION WATCHER
-- ═══════════════════════════════════════════════════════════
local function watchNotifications()
    task.spawn(function()
        while true do
            task.wait(0.3)
            pcall(function()
                local containers = {
                    CoreGui:FindFirstChild("NotificationRoot"),
                    parent:FindFirstChild("NotificationRoot"),
                }
                for _, nc in ipairs(containers) do
                    if nc then
                        for _, v in ipairs(nc:GetDescendants()) do
                            if v:IsA("TextLabel") and v.Text and v.Text ~= "" then
                                local text = v.Text
                                local rankName = text:match("tempRanked to '([^']+)'")
                                    or text:match("permRanked to '([^']+)'")
                                    or text:match("ranked to '([^']+)'")
                                    or text:match("you are now (%a+)")
                                local rankLevel = text:match("%((%d+)%)")
                                if rankName then
                                    rankName = rankName:gsub("!$", "")
                                    local lvl = tonumber(rankLevel) or nameToLevel(rankName)
                                    if lvl then setAchieved(lvl, "notif") end
                                end
                            end
                        end
                    end
                end
            end)
        end
    end)
end

-- ═══════════════════════════════════════════════════════════
-- ESCALATION — 3 phases
-- ═══════════════════════════════════════════════════════════
local function escalate()
    NR.addLog("═══════════════════════════════════════")
    NR.addLog("ESCALATION START")
    NR.addLog("═══════════════════════════════════════")
    
    -- Initial detect
    detect()
    task.wait(0.5)
    NR.addLog("initial: " .. State.currentName .. " (" .. State.currentRank .. ")")
    
    -- Phase 1: Chat
    NR.addLog("--- PHASE 1: Chat commands ---")
    for i, cmd in ipairs(CONFIG.CHAT_ESCALATION) do
        NR.addLog(string.format("[%d/%d] %s", i, #CONFIG.CHAT_ESCALATION, cmd))
        local before = State.achievedRank
        NR.sendChat(cmd)
        task.wait(CONFIG.SETTLE)
        detect()
        if State.achievedRank > before then
            NR.addLog(">>> HIT via chat: " .. cmd)
            if State.achievedRank >= 4 then
                NR.addLog("reached Admin+ — done")
                return State.achievedRank
            end
        end
    end
    
    -- Phase 2: Remotes
    NR.addLog("--- PHASE 2: Remote payloads ---")
    local remotes = collectRemotes()
    local targets = {}
    for _, r in ipairs(remotes.suspects) do table.insert(targets, r) end
    for _, r in ipairs(remotes.nonHD) do
        local dup = false
        for _, t in ipairs(targets) do if t == r then dup = true break end end
        if not dup then table.insert(targets, r) end
    end
    NR.addLog(string.format("targets: %d (suspects: %d)", #targets, #remotes.suspects))
    
    for i, p in ipairs(CONFIG.PAYLOADS) do
        NR.addLog(string.format("[%d/%d] %s", i, #CONFIG.PAYLOADS, p.label))
        local before = State.achievedRank
        fireParallel(targets, p.args)
        task.wait(CONFIG.SETTLE)
        detect()
        if State.achievedRank > before then
            State.hitPayload = p
            NR.addLog(">>> HIT: " .. p.label)
            if State.achievedRank >= 4 then
                NR.addLog("reached Admin+ — done")
                return State.achievedRank
            end
        end
    end
    
    -- Phase 3: Race
    NR.addLog("--- PHASE 3: Race condition ---")
    local RACE = {
        {"SetRank", LP.UserId, "Owner", "Perm"},
        {"SetRank", LP.UserId, 5, "Perm"},
        {"buyAdmin", "Owner", 0},
        {"RequestRank", LP.UserId, 255},
    }
    for _, args in ipairs(RACE) do
        raceFire(remotes.suspects, args)
        task.wait(1.5)
    end
    task.wait(CONFIG.SETTLE)
    detect()
    
    NR.addLog("═══════════════════════════════════════")
    NR.addLog(string.format("DONE — highest: %s (%d) via %s",
        State.achievedName, State.achievedRank, State.hitSource or "—"))

end

local function inject()
    if State.injected then return end
    NR.addLog("injecting")
    
    task.spawn(function()
        setProgress(0.05, "Initializing...")
        task.wait(0.3)
        
        setProgress(0.15, "Detecting rank...")
        detect()
        task.wait(0.3)
        
        setProgress(0.25, "Collecting remotes...")
        collectRemotes()
        task.wait(0.3)
        
        if CONFIG.AUTO_ESCALATE then
            setProgress(0.4, "Escalating...")
            escalate()
        end
        
        setProgress(0.85, "Building UI...")
        task.wait(0.3)
        NR.generateButtons()
        
        setProgress(1.0, "Complete")
        task.wait(0.4)
        
        State.injected = true
        overlay.Visible = false
        mainFrame.Visible = true
        cmdInputRow.Visible = true
        
        watchNotifications()
        
        NR.notify("NullRoot", string.format("Ready — %s (%d)", State.currentName, State.currentRank), 4)
        NR.addLog("Ready — " .. State.currentName .. " (" .. State.currentRank .. ") src: " .. (State.hitSource or "—"))
    end)
